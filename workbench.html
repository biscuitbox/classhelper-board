// =====================================================================
//  helper.h  —  모든 앱이 함께 쓰는 것: 화면(lcd), 색, 단추 그리기, '앱 약속'
//
//  ┌───────────────────────── 앱 약속 ─────────────────────────┐
//  │ 앱 하나 = 파일 하나 (app_mine.h / app_noise.h / app_presenter.h) │
//  │ 파일 안은 namespace 하나로 감싸고, 아래 4개 함수를 만듭니다.      │
//  │                                                                │
//  │   void start()   ① 앱에 들어올 때 — 화면 전체를 처음부터 그린다   │
//  │   void update()  ② 계속 반복(1초에 약 100번) — 시계·센서·애니메이션 │
//  │   void touch(uint8_t ev, int x, int y)                         │
//  │                  ③ 터치 — ev 는 TOUCH_DOWN / TOUCH_MOVE / TOUCH_UP │
//  │   void stop()    ④ 나갈 때 — 마이크 끄기, 키 떼기 같은 정리       │
//  │                                                                │
//  │ 파일 맨 끝에 앱 정보(App)를 하나 적습니다. 예)                   │
//  │   const App app = { "지뢰찾기", "심심할 때 한 판", C_MINE, icon,  │
//  │                     start, update, touch, stop,                │
//  │                     nullptr,   // BOOT 버튼 (안 쓰면 nullptr)   │
//  │                     false,     // true = 이 앱에서는 절전 안 함   │
//  │                     true };    // true = 완성 (false = 회색 칸)   │
//  │                                                                │
//  │ 화면 규칙                                                       │
//  │  - 화면은 세로 240 x 320. 왼쪽 위 (0,0)~(43,29) 는 🏠 자리.      │
//  │    start() 에서 drawHomeButton() 을 꼭 불러 주세요.             │
//  │    이 자리의 터치는 앱으로 오지 않습니다 (길게 누르면 첫 화면).   │
//  │  - 글자: 한글은 fonts::efontKR_12 / 16 / 24 (굵게는 _b)          │
//  │  - 이모지는 화면에 안 나옵니다. 아이콘은 도형으로 그리세요.       │
//  │  - delay() 를 길게 쓰지 마세요 (터치가 멈춤). millis() 로 시간 재기. │
//  │                                                                │
//  │ 함께 쓸 수 있는 도구                                             │
//  │  - 소리 : Sfx::play(SFX_TAP)   (sfx.h 에 소리 목록)              │
//  │  - 마이크 : Mic::start() / Mic::hasNew() / Mic::db() / Mic::stop() │
//  │  - 블루투스 키보드·마우스 : Hid::key(...) 등 (ble_hid.h 참고)    │
//  │  - 저장 : Preferences (앱마다 다른 이름표로 prefs.begin("이름"))   │
//  └────────────────────────────────────────────────────────────────┘
// =====================================================================

enum : uint8_t { LED_OFF, LED_FLASH, LED_BOOM, LED_WIN };
static struct { uint8_t fx = LED_OFF; uint32_t t0 = 0, dur = 0; uint8_t r = 0, g = 0, b = 0; uint32_t last = 0; } led;

static void ledStart(uint8_t fx, uint32_t dur, uint8_t r = 0, uint8_t gg = 0, uint8_t b = 0) {
  led.fx = fx; led.t0 = millis(); led.dur = dur; led.r = r; led.g = gg; led.b = b;
}
static void ledUpdate() {
  if (led.fx == LED_OFF) return;
  uint32_t now = millis();
  if (now - led.last < 20) return;
  led.last = now;
  uint32_t t = now - led.t0;
  if (t >= led.dur) { rgbLedWrite(PIN_WS2812, 0, 0, 0); led.fx = LED_OFF; return; }
  float k = 1.0f - (float)t / led.dur;
  if (led.fx == LED_FLASH) {
    rgbLedWrite(PIN_WS2812, led.r * k, led.g * k, led.b * k);
  } else if (led.fx == LED_BOOM) {
    float p = 0.5f + 0.5f * cosf(t / 55.0f);
    rgbLedWrite(PIN_WS2812, 90 * k * p + 10 * k, 12 * k * p, 0);
  } else {                                   // WIN: 무지개
    float h = fmodf(t / 300.0f, 1.0f) * 6.0f;
    int   i = (int)h; float f = h - i;
    float r = 0, gg = 0, b = 0;
    switch (i) {
      case 0: r = 1; gg = f; break;      case 1: r = 1 - f; gg = 1; break;
      case 2: gg = 1; b = f; break;      case 3: gg = 1 - f; b = 1; break;
      case 4: r = f; b = 1; break;       default: r = 1; b = 1 - f;
    }
    rgbLedWrite(PIN_WS2812, r * 60 * k, gg * 60 * k, b * 60 * k);
  }
}

static void setupLevel(uint8_t lv) {
  const Level& L = LEVELS[lv];
  g.level = lv; g.cols = L.cols; g.rows = L.rows; g.mines = L.mines; g.cell = L.cell;
  g.bx = (240 - g.cols * g.cell) / 2;
  g.by = BOARD_Y + (BOARD_H - g.rows * g.cell) / 2;
}

static void computeAdj() {
  for (int r = 0; r < g.rows; r++)
    for (int c = 0; c < g.cols; c++) {
      int n = 0;
      for (int dr = -1; dr <= 1; dr++)
        for (int dc = -1; dc <= 1; dc++) {
          int rr = r + dr, cc = c + dc;
          if ((dr || dc) && rr >= 0 && rr < g.rows && cc >= 0 && cc < g.cols) n += g.mine[rr * g.cols + cc];
        }
      g.adj[r * g.cols + c] = n;
    }
}

static void placeMines(int safe) {
  int sc = safe % g.cols, sr = safe / g.cols, n = g.cols * g.rows, placed = 0;
  while (placed < g.mines) {
    int i = esp_random() % n;
    int c = i % g.cols, r = i / g.cols;
    if (g.mine[i] || (abs(c - sc) <= 1 && abs(r - sr) <= 1)) continue;   // 첫 칸 주변은 항상 안전
    g.mine[i] = 1; placed++;
  }
  computeAdj();
}

  memset(g.adj, 0, sizeof(g.adj));

  g.status = G_READY; g.opened = 0; g.flags = 0; g.elapsedBase = 0; g.boomCell = -1;
  started = true;
  drawGameScreen();
  if (sound) Sfx::play(SFX_START);
}

// 연쇄 열기 (빈칸이면 주변까지). 연 칸 수를 돌려줌
static int floodOpen(int start) {
  static int16_t stack[MAX_CELLS];
  if (g.st[start] != HIDDEN) return 0;
  int sp = 0, count = 0;
  g.st[start] = OPEN; g.opened++; count++; drawCell(start, false);
  if (g.adj[start] == 0) stack[sp++] = start;
  while (sp) {
    int i = stack[--sp], c = i % g.cols, r = i / g.cols;
    for (int dr = -1; dr <= 1; dr++)
      for (int dc = -1; dc <= 1; dc++) {
        int rr = r + dr, cc = c + dc;
        if ((!dr && !dc) || rr < 0 || rr >= g.rows || cc < 0 || cc >= g.cols) continue;
        int j = rr * g.cols + cc;
        if (g.st[j] != HIDDEN || g.mine[j]) continue;
        g.st[j] = OPEN; g.opened++; count++; drawCell(j, false);
        if (g.adj[j] == 0) stack[sp++] = j;
      }
  }
  return count;
}

static void lose(int i) {
  g.elapsedBase = elapsedSec();
  g.status = G_LOST; g.boomCell = i; g.endMs = millis();
  lcd.startWrite();
  for (int k = 0; k < g.cols * g.rows; k++)

  drawFace(FACE_DEAD);
  lcd.endWrite();
  Sfx::play(SFX_BOOM);
  ledStart(LED_BOOM, 1400);
}

static void checkWin() {
  if (g.status != G_PLAY || g.opened != g.cols * g.rows - g.mines) return;
  g.elapsedBase = elapsedSec();
  g.status = G_WON; g.endMs = millis();
  lcd.startWrite();
  for (int k = 0; k < g.cols * g.rows; k++)
    if (g.mine[k] && g.st[k] != FLAG) { g.st[k] = FLAG; drawCell(k, false); }
  g.flags = g.mines;
  drawMineCounter();
  drawTimer();
  drawFace(FACE_COOL);
  uint16_t t = g.elapsedBase;
  bool record = (t < best[g.level]);
  if (record) { best[g.level] = t; savePrefs(); }
  char l2[40];
  snprintf(l2, sizeof(l2), record ? "%u초 · 신기록!" : "%u초 걸렸어요", (unsigned)t);
  drawBanner(record ? "축하해요! 성공!" : "성공!", l2);
  lcd.endWrite();
  Sfx::play(record ? SFX_RECORD : SFX_WIN);
  ledStart(LED_WIN, 2600);
}

static void doReveal(int i) {
  if (g.st[i] != HIDDEN) return;
  if (g.status == G_READY) {
    placeMines(i);
    g.status = G_PLAY; g.startMs = millis(); g.elapsedBase = 0;
  }
  if (g.mine[i]) { lose(i); return; }
  lcd.startWrite();
  int n = floodOpen(i);
  lcd.endWrite();
  Sfx::play(n > 1 ? SFX_FLOOD : SFX_OPEN);
  checkWin();
}

static void doChord(int i) {
  if (g.st[i] != OPEN || g.adj[i] == 0 || g.status != G_PLAY) return;
  int c = i % g.cols, r = i / g.cols, fl = 0, hidden = 0;
  for (int dr = -1; dr <= 1; dr++)
    for (int dc = -1; dc <= 1; dc++) {
      int rr = r + dr, cc = c + dc;
      if ((!dr && !dc) || rr < 0 || rr >= g.rows || cc < 0 || cc >= g.cols) continue;
      uint8_t s = g.st[rr * g.cols + cc];
      if (s == FLAG) fl++; else if (s == HIDDEN) hidden++;
    }
  if (hidden == 0) return;
  if (fl != g.adj[i]) { Sfx::play(SFX_ERROR); return; }
  int boom = -1, total = 0;
  lcd.startWrite();
  for (int dr = -1; dr <= 1; dr++)
    for (int dc = -1; dc <= 1; dc++) {
      int rr = r + dr, cc = c + dc;
      if ((!dr && !dc) || rr < 0 || rr >= g.rows || cc < 0 || cc >= g.cols) continue;
      int j = rr * g.cols + cc;
      if (g.st[j] != HIDDEN) continue;
      if (g.mine[j]) boom = j; else total += floodOpen(j);
    }
  lcd.endWrite();
  if (boom >= 0) { lose(boom); return; }
  if (total) Sfx::play(SFX_CHORD);
  checkWin();
}

static void toggleFlag(int i) {
  if (g.status == G_WON || g.status == G_LOST) return;
  if (g.st[i] == HIDDEN)    { g.st[i] = FLAG;   g.flags++; Sfx::play(SFX_FLAG); ledStart(LED_FLASH, 250, 0, 20, 70); }
  else if (g.st[i] == FLAG) { g.st[i] = HIDDEN; g.flags--; Sfx::play(SFX_UNFLAG); }
  else return;
  lcd.startWrite();
  drawCell(i, false);
  drawMineCounter();
  lcd.endWrite();
}

static struct {
  uint32_t t0 = 0; uint8_t target = T_NONE; int16_t cell = -1; bool consumed = false;
} tp;

static int16_t cellAt(int x, int y) {
  const int tol = 6;                         // 판 가장자리 살짝 밖도 허용
  int w = g.cols * g.cell, h = g.rows * g.cell;
  if (x < g.bx - tol || x >= g.bx + w + tol || y < g.by - tol || y >= g.by + h + tol) return -1;
  int c = constrain((x - g.bx) / g.cell, 0, g.cols - 1);
  int r = constrain((y - g.by) / g.cell, 0, g.rows - 1);
  if (x < g.bx) c = 0;
  if (y < g.by) r = 0;
  return r * g.cols + c;
}

static bool gameActive() { return g.status == G_READY || g.status == G_PLAY; }

// 누르고 있는 동안 '눌린 칸' 표시
static int16_t pressedList[9];
static uint8_t pressedN = 0;

static void setPressedFor(int16_t cell) {
  int16_t nl[9]; uint8_t nn = 0;
  if (cell >= 0 && gameActive()) {
    if (g.st[cell] == HIDDEN) nl[nn++] = cell;
    else if (g.st[cell] == OPEN && g.adj[cell]) {
      int c = cell % g.cols, r = cell / g.cols;
      for (int dr = -1; dr <= 1; dr++)
        for (int dc = -1; dc <= 1; dc++) {
          int rr = r + dr, cc = c + dc;
          if ((!dr && !dc) || rr < 0 || rr >= g.rows || cc < 0 || cc >= g.cols) continue;
          int j = rr * g.cols + cc;
          if (g.st[j] == HIDDEN) nl[nn++] = j;
        }
    }
  }
  lcd.startWrite();
  for (int a = 0; a < pressedN; a++) {
    bool keep = false;
    for (int b = 0; b < nn; b++) if (nl[b] == pressedList[a]) keep = true;
    if (!keep) drawCell(pressedList[a], false);
  }
  for (int b = 0; b < nn; b++) {
    bool had = false;
    for (int a = 0; a < pressedN; a++) if (pressedList[a] == nl[b]) had = true;
    if (!had) drawCell(nl[b], true);
  }
  lcd.endWrite();
  memcpy(pressedList, nl, sizeof(nl));
  pressedN = nn;
}
static void clearPressed() { setPressedFor(-1); }

static void shortAction(int i) {
  if (!flagMode) {
    if (g.st[i] == HIDDEN) doReveal(i);
    else if (g.st[i] == OPEN) doChord(i);
  } else {
    if (g.st[i] != OPEN) toggleFlag(i);
    else doChord(i);
  }
}
static void longAction(int i) {
  if (!flagMode) {
    if (g.st[i] != OPEN) toggleFlag(i);
    else doChord(i);
  } else {
    if (g.st[i] == HIDDEN) doReveal(i);
    else if (g.st[i] == FLAG) toggleFlag(i);
    else doChord(i);
  }
}

static void gameDown(int x, int y) {
  tp.consumed = false; tp.t0 = millis(); tp.cell = -1;
  if (y < TOP_H) {
    if (x >= 98 && x <= 142) { tp.target = T_FACE; drawFace(curFace(), true); }
    else tp.target = T_NONE;
  } else if (y >= BOT_Y - 4) {
    if (x < 110)       { tp.target = T_MODE; drawModeButton(true); }
    else if (x >= 176) { tp.target = T_MENU; drawMenuButton(true); }
    else tp.target = T_NONE;
  } else {
    tp.target = T_BOARD;
    tp.cell = cellAt(x, y);
    if (gameActive()) { setPressedFor(tp.cell); drawFace(FACE_OH); }
  }
}

static void gameMove(int x, int y) {
  if (tp.target != T_BOARD) return;
  int16_t c = cellAt(x, y);
  if (c != tp.cell) {
    tp.cell = c; tp.t0 = millis();
    if (!tp.consumed && gameActive()) setPressedFor(c);
  }
}

static void gameHold() {
  if (tp.target == T_BOARD && !tp.consumed && gameActive() && tp.cell >= 0 &&
      millis() - tp.t0 >= LONG_PRESS_MS) {
    tp.consumed = true;
    clearPressed();
    longAction(tp.cell);
    if (gameActive()) drawFace(FACE_SMILE);
  }
}

static void gameUp(int x, int y) {
  switch (tp.target) {
    case T_FACE:
      newGame(level);
      break;

static void startCalibration() {
  calibrating = true;
  calStart = millis() + 400;              // 단추 소리가 끝난 뒤부터 재기
  calSum = 0; calN = 0;
  drawCalibration(3);
}

static void finishCalibration() {
  calibrating = false;
  if (calN > 5) { baseDb = (float)(calSum / calN); hasBase = true; savePrefs(); }
  level = 0; state = cand = GREEN; candSince = millis();
  drawPanel();
  Sfx::play(SFX_FLAG);
}

// 새 측정값 하나 처리
static void onSample(float db) {
  uint32_t now = millis();
  if (calibrating) {
    if ((int32_t)(now - calStart) >= 0) { calSum += db; calN++; }
    return;
  }
  float rel = constrain(db - baseDb, 0.0f, SCALE_MAX);
  // 커질 때는 빨리, 작아질 때는 천천히 따라가기 (깜빡임 방지)
  level += (rel - level) * (rel > level ? 0.45f : 0.08f);

  uint8_t target = level >= THRESH[mode][1] ? RED : (level >= THRESH[mode][0] ? YELLOW : GREEN);
  if (target != cand) { cand = target; candSince = now; }
  uint32_t need = (cand > state) ? 600 : 1500;        // 나빠질 땐 0.6초, 좋아질 땐 1.5초 유지돼야 바뀜
  if (cand != state && now - candSince >= need) {
    state = cand;
    if (state == RED) redSince = now;
    drawPanel();
  }
  // 경고음
  if (beepOn && state == RED && now - redSince >= RED_BEEP_MS && now - lastBeep >= BEEP_COOLDOWN_MS) {
    Sfx::play(SFX_ALERT);
    lastBeep = now;
  }
  if (level > secMax) secMax = level;
}

static void start() {
  static bool loaded = false;
  if (!loaded) {
    loadPrefs();
    meter.createSprite(240, 42);
    graph.createSprite(240, 42);
    memset(hist, 0, sizeof(hist));
    loaded = true;
  }
  level = 0; state = cand = GREEN; candSince = millis();
  secMax = 0; secStart = millis();
  pressedBtn = -1;
  Mic::start();
  calibrating = false;
  drawAll();
  if (!hasBase) startCalibration();       // 처음 쓸 때는 기준부터 맞추기
}

static void update() {
  uint32_t now = millis();
  while (Mic::hasNew()) onSample(Mic::db());

  if (calibrating) {
    static int shown = -1;
    int left = 3 - (int)((int32_t)(now - calStart) / 1000);
    if (left < 1) left = 1;
    if (left != shown) { shown = left; drawCalibration(left); }
    if ((int32_t)(now - calStart) >= CAL_MS) { shown = -1; finishCalibration(); }
    return;
  }

  static uint32_t lastMeter = 0;
  if (now - lastMeter >= 80) { lastMeter = now; drawMeter(); }

  if (now - secStart >= 1000) {           // 1초마다 그래프에 한 칸
    secStart = now;
    hist[histPos] = (uint8_t)(secMax + 0.5f);
    histPos = (histPos + 1) % 60;
    secMax = 0;
    drawGraph();
  }
}

static int buttonAt(int x, int y) {
  if (y < 284) return -1;
  if (x < 4 + 3 * 58) return constrain((x - 4) / 58, 0, 2);
  return 3;
}

static void touch(uint8_t ev, int x, int y) {
  if (ev == TOUCH_DOWN) {
    pressedBtn = buttonAt(x, y);
    if (pressedBtn >= 0 && pressedBtn < 3) buttonPressed(4 + pressedBtn * 58, 286, 54, 30);
    if (pressedBtn == 3) buttonPressed(178, 286, 58, 30);
    return;
  }
  if (ev != TOUCH_UP) return;

  if (inside(x, y, 166, 0, 74, 30)) {                    // 경고음 켜고 끄기
    beepOn = !beepOn; savePrefs(); drawBeepButton();
    Sfx::play(SFX_TAP);
    return;
  }
  int b = buttonAt(x, y);
  int was = pressedBtn;
  pressedBtn = -1;
  if (b < 0 || b != was) { drawButtons(); return; }
  Sfx::play(SFX_TAP);
  if (b < 3) {
    mode = b; savePrefs();
    drawButtons();
    if (!calibrating) drawPanel();
    drawGraph();
  } else {
    drawButtons();
    startCalibration();
  }
}

static void stop() {
  Mic::stop();                            // ⚠️ 나갈 때 마이크 끄기
  savePrefs();
}

static void drawBtn(int i, bool down = false) {
  const Btn& b = btns[i];
  uint16_t face = !b.enabled ? 0x8410 : (b.act == tool && tab == TAB_POINTER && b.act >= A_LASER && b.act <= A_ARROW)
                                       ? C_ACCENT : C_WHITE;
  uint16_t ink  = (face == C_WHITE) ? C_BLACK : C_WHITE;
  if ((b.act == A_PREV || b.act == A_NEXT) && b.h >= 100) {    // 큰 화살표 단추
    face = b.enabled ? (b.act == A_NEXT ? C_PRES : 0x5AEB) : 0x8410;
    lcd.fillRoundRect(b.x, b.y, b.w, b.h, 12, face);
    int cx = b.x + b.w / 2, cy = b.y + b.h / 2 - 8, s = 26;
    if (b.act == A_NEXT) lcd.fillTriangle(cx - s * 2 / 3, cy - s, cx - s * 2 / 3, cy + s, cx + s, cy, C_WHITE);
    else                 lcd.fillTriangle(cx + s * 2 / 3, cy - s, cx + s * 2 / 3, cy + s, cx - s, cy, C_WHITE);
    text(b.label, cx, b.y + b.h - 18, &fonts::efontKR_16_b, C_WHITE);
  } else {
    lcd.fillRoundRect(b.x, b.y, b.w, b.h, 8, face);
    lcd.drawRoundRect(b.x, b.y, b.w, b.h, 8, C_DARK);
    text(b.label, b.x + b.w / 2, b.y + b.h / 2 + 1,
         (b.h >= 44) ? &fonts::efontKR_16_b : &fonts::efontKR_14, ink);
  }

    lcd.drawRoundRect(b.x + 1, b.y + 1, b.w - 2, b.h - 2, 7, C_YELLOW);

  }
}

static void sendAction(uint8_t a) {
  switch (a) {
    // ① 넘기기
    case A_PREV:        Hid::key(KEY_LEFT_ARROW);  break;
    case A_NEXT:        Hid::key(KEY_RIGHT_ARROW); break;
    case A_START_FIRST: if (google) Hid::combo3(KEY_LEFT_CTRL, KEY_LEFT_SHIFT, KEY_F5);
                        else        Hid::key(KEY_F5);                         break;
    case A_START_HERE:  if (google) Hid::combo(KEY_LEFT_CTRL, KEY_F5);
                        else        Hid::combo(KEY_LEFT_SHIFT, KEY_F5);       break;
    case A_END:         Hid::key(KEY_ESC); break;

    // ② 포인터 (PPT 단축키: Ctrl+L 레이저 / Ctrl+P 펜 / Ctrl+E 지우개 / Ctrl+A 화살표 / E 다 지우기)
    case A_LASER:  if (google) Hid::key('l'); else Hid::combo(KEY_LEFT_CTRL, 'l');
                   tool = (tool == A_LASER && google) ? A_ARROW : A_LASER; break;
    case A_PEN:    Hid::combo(KEY_LEFT_CTRL, 'p'); tool = A_PEN;    break;
    case A_ERASER: Hid::combo(KEY_LEFT_CTRL, 'e'); tool = A_ERASER; break;
    case A_ARROW:  Hid::combo(KEY_LEFT_CTRL, 'a'); tool = A_ARROW;  break;
    case A_ERASE_ALL: Hid::key('e'); break;

    // ③ 화면·소리  (PPT 는 한/영 상태와 상관없는 '.' ',' 사용)
    case A_BLACK:  Hid::key(google ? 'b' : '.'); break;
    case A_WHITE:  Hid::key(google ? 'w' : ','); break;
    case A_VOL_DOWN: Hid::media(CONSUMER_CONTROL_VOLUME_DECREMENT); break;
    case A_MUTE:     Hid::media(CONSUMER_CONTROL_MUTE);             break;
    case A_VOL_UP:   Hid::media(CONSUMER_CONTROL_VOLUME_INCREMENT); break;
    case A_VIDEO:       Hid::combo(KEY_LEFT_ALT, 'p');  break;
    case A_HIDE_CURSOR: Hid::combo(KEY_LEFT_CTRL, 'h'); break;
  }
}

static void onButton(uint8_t a) {
  // 번호 키패드
  if (a >= A_K0 && a <= A_K9) {
    if (keypadNum < 1000) keypadNum = keypadNum * 10 + (a - A_K0);
    Sfx::play(SFX_KEY); drawContent(); return;
  }
  switch (a) {
    case A_KEYPAD:  keypadOn = true; keypadNum = 0; Sfx::play(SFX_TAP); drawContent(); return;
    case A_K_CLOSE: keypadOn = false; Sfx::play(SFX_TAP); drawContent(); return;
    case A_K_BACK:  keypadNum /= 10; Sfx::play(SFX_KEY); drawContent(); return;
    case A_K_GO:
      if (keypadNum > 0 && hidReady()) { Hid::typeNumber(keypadNum); Sfx::play(SFX_KEY); }
      else Sfx::play(SFX_ERROR);
      keypadOn = false; drawContent(); return;

    // ④ 타이머 (PC 연결 없이도 작동)
    case A_T_PRESET0: case A_T_PRESET1: case A_T_PRESET2: case A_T_PRESET3:
      timerSet(PRESET_MIN[a - A_T_PRESET0] * 60000UL); Sfx::play(SFX_TAP); drawContent(); return;
    case A_T_MINUS: case A_T_PLUS: {
      int32_t d = (a == A_T_PLUS) ? 60000 : -60000;
      if (tRunning) { if (d > 0 || timerLeft() > 60000) { tEndAt += d; tTotal = max<int32_t>(60000, (int32_t)tTotal + d); } }
      else { int32_t v = (int32_t)tLeftPaused + d; if (v >= 60000 && v <= 99 * 60000) { tLeftPaused = v; tTotal = v; } }
      Sfx::play(SFX_TAP); drawTimerFace(); return;
    }
    case A_T_TOGGLE:
      if (tRunning) { tLeftPaused = timerLeft(); tRunning = false; }
      else if (tLeftPaused > 0) { tEndAt = millis() + tLeftPaused; tRunning = true; tDone = false; }
      Sfx::play(SFX_TAP); drawContent(); drawTabs(); return;
    case A_T_RESET:
      tRunning = false; tLeftPaused = tTotal; tDone = false;
      Sfx::play(SFX_TAP); drawContent(); drawTabs(); return;
  }
  // 나머지 = PC 로 키 보내기
  if (!hidReady()) { Sfx::play(SFX_ERROR); toast("블루투스 연결부터 해 주세요", C_STOP); delay(600); drawContent(); return; }
  uint8_t oldTool = tool;
  sendAction(a);
  Sfx::play(SFX_KEY);
  if (tool != oldTool) drawContent();           // 도구 바뀐 것 표시
}

static struct { bool on = false; int lx = 0, ly = 0; float ax = 0, ay = 0; uint32_t t0 = 0; int dist = 0; bool drag = false; } pad;
static bool inPad(int x, int y) { return tab == TAB_POINTER && Hid::connected() && inside(x, y, 6, 78, 228, 158); }

static void padDown(int x, int y) {
  pad.on = true; pad.lx = x; pad.ly = y; pad.ax = pad.ay = 0; pad.t0 = millis(); pad.dist = 0;
  pad.drag = (tool == A_PEN || tool == A_ERASER);
  if (pad.drag && hidReady()) Hid::mouseDown();          // 펜·지우개: 누른 채로 움직이면 그려짐
}
static void padMove(int x, int y) {
  int dx = x - pad.lx, dy = y - pad.ly;
  pad.lx = x; pad.ly = y;
  pad.dist += abs(dx) + abs(dy);
  pad.ax += dx * PAD_SPEED; pad.ay += dy * PAD_SPEED;
  int mx = (int)pad.ax, my = (int)pad.ay;
  pad.ax -= mx; pad.ay -= my;
  if (hidReady()) Hid::mouseMove(mx, my);
}
static void padUp() {
  pad.on = false;
  if (!hidReady()) return;
  if (pad.drag) Hid::mouseUp();
  else if (pad.dist < 8 && millis() - pad.t0 < 300) { Hid::click(); Sfx::play(SFX_KEY); }
}

static void start() {
  static bool loaded = false;
  if (!loaded) { loadPrefs(); loaded = true; }
  pressed = -1; pad.on = false; keypadOn = false;
  drawAll();
}

static void update() {
  uint32_t now = millis();
  // PC 연결 상태 점
  static bool lastConn = false;
  bool c = Hid::connected();
  if (c != lastConn) {                       // 연결되거나 끊기면 화면을 다시 그림
    lastConn = c;
    drawConnDot();
    if (tab != TAB_TIMER) { keypadOn = false; pressed = -1; pad.on = false; drawContent(); }
    Sfx::play(c ? SFX_FLAG : SFX_UNFLAG);
  }
  static uint32_t lastBat = 0;
  if (now - lastBat > 15000) { lastBat = now; drawConnDot(); }

  // 타이머
  if (tRunning && timerLeft() == 0) {
    tRunning = false; tDone = true; tDoneAt = now; tLeftPaused = 0;
    Sfx::play(SFX_ALARM);
    tab = TAB_TIMER; keypadOn = false;           // 어느 장에 있든 타이머 장으로
    drawTabs(); drawContent();
  }
  static uint32_t lastTick = 0;
  if (now - lastTick >= 250) {
    lastTick = now;
    if (tDone) {
      bool on = ((now - tDoneAt) / 500) % 2 == 0;
      static bool lastOn = true;
      if (tab == TAB_TIMER && on != lastOn) { lastOn = on; lcd.startWrite(); drawTimerDone(on); lcd.endWrite(); }
      if (now - tDoneAt > 60000) { tDone = false; tLeftPaused = tTotal; if (tab == TAB_TIMER) drawContent(); }
    } else if (tRunning) {
      static uint32_t lastSec = 0;
      uint32_t s = timerLeft() / 1000;
      if (s != lastSec) {
        lastSec = s;
        if (tab == TAB_TIMER) drawTimerFace(); else drawTabs();
      }
    }
  }
}

static int btnAt(int x, int y) {
  for (int i = 0; i < nb; i++)
    if (inside(x, y, btns[i].x, btns[i].y, btns[i].w, btns[i].h)) return i;
  return -1;
}

static void touch(uint8_t ev, int x, int y) {
  if (ev == TOUCH_DOWN) {
    if (tab == TAB_TIMER && tDone) {             // 시간 끝 화면 → 누르면 멈춤
      tDone = false; tLeftPaused = tTotal; Sfx::play(SFX_TAP); drawContent(); return;
    }
    if (inPad(x, y)) { padDown(x, y); return; }
    pressed = btnAt(x, y);
    if (pressed >= 0 && btns[pressed].enabled) drawBtn(pressed, true);
    return;
  }
  if (ev == TOUCH_MOVE) {
    if (pad.on) padMove(x, y);
    return;
  }
  // TOUCH_UP
  if (pad.on) { padUp(); return; }

  if (y < 30) {                                  // 제목줄
    if (x >= 168) { google = !google; savePrefs(); tool = A_ARROW; Sfx::play(SFX_TAP); drawModeToggle(); drawContent(); }
    return;
  }
  if (y >= 278) {                                // 아래 탭
    uint8_t t = constrain(x / 60, 0, 3);
    if (t != tab) { tab = t; keypadOn = false; Sfx::play(SFX_TAP); drawTabs(); drawContent(); }
    pressed = -1;
    return;
  }
  int i = pressed;
  pressed = -1;
  if (i < 0 || i >= nb) return;
  drawBtn(i);
  if (!btns[i].enabled) { Sfx::play(SFX_ERROR); return; }
  if (btnAt(x, y) != i) return;                  // 손가락을 단추 밖으로 빼면 취소
  onButton(btns[i].act);
}

static void stop() {
  if (pad.on && pad.drag && hidReady()) Hid::mouseUp();   // 눌린 채로 남지 않게
  pad.on = false;
  savePrefs();
}

// 보드 뒷면 BOOT 단추 = 다음 장
static void bootButton() {
  if (hidReady()) { Hid::key(KEY_RIGHT_ARROW); Sfx::play(SFX_KEY); }
  else Sfx::play(SFX_ERROR);
}

const App app = {
  "프레젠터", "PPT 넘기기·포인터·타이머", C_PRES, icon,
  start, update, touch, stop,
  bootButton,  // BOOT 버튼 = 다음 장
  true,        // 수업 중 화면이 어두워지지 않게
  true         // 완성!
};
